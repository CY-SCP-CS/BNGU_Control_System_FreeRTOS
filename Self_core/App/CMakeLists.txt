# Each module exports its public headers only; private headers are compiler-private.
function(app_module name kind)
    set(target app_${name}_${kind})
    add_library(${target} OBJECT ${ARGN})
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/internal")
    target_link_libraries(${target} PRIVATE self_app_contracts self_config self_lib)
    if(NOT kind STREQUAL "algorithm")
        target_link_libraries(${target} PRIVATE self_os_api)
    endif()
    if(kind STREQUAL "runtime")
        target_link_libraries(${target} PRIVATE self_os_api self_drv_types)
    elseif(kind STREQUAL "port")
        target_link_libraries(${target} PRIVATE self_os_api self_drv self_bsp)
    endif()
    set_property(GLOBAL APPEND PROPERTY SELF_OBJECT_TARGETS ${target})
endfunction()

    control_support navigation armored ui)
foreach(module IN LISTS app_modules)
    add_library(app_${module}_api INTERFACE)
    target_link_libraries(app_${module}_api INTERFACE self_app_contracts)
    if(DEFINED vehicle_path_${module})
        target_include_directories(app_${module}_api INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/Vehicles/${vehicle_path_${module}}")
    else()
        target_include_directories(app_${module}_api INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/Modules/${module}")
    endif()
endforeach()

    if(NOT DEFINED vehicle_path_${module})
        add_subdirectory(Modules/${module})
    endif()
endforeach()
