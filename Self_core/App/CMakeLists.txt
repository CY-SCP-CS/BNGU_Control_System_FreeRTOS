# Modules own source inventories; this file only composes them.
function(app_module name kind)
    set(target app_${name}_${kind})
    add_library(${target} OBJECT ${ARGN})
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/internal")
    target_link_libraries(${target} PRIVATE self_app_contracts self_config self_lib self_app)
    if(NOT kind STREQUAL "algorithm")
        target_link_libraries(${target} PRIVATE self_app_tasks_api)
    endif()
    target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/Self_core/App/Modules/control_support")
    if(name STREQUAL "dbus" OR name STREQUAL "imu" OR name STREQUAL "referee")
        target_link_libraries(${target} PRIVATE self_drv_types)
    endif()
    if(kind STREQUAL "runtime")
        target_link_libraries(${target} PRIVATE self_os self_drv_types)
    elseif(kind STREQUAL "port")
        target_link_libraries(${target} PRIVATE self_os self_drv self_bsp)
    endif()
    set_property(GLOBAL APPEND PROPERTY SELF_OBJECT_TARGETS ${target})
endfunction()

# The algorithm include set contains business contracts and pure module APIs only.
add_library(self_app INTERFACE)
add_library(self_app_tasks INTERFACE)
add_library(self_app_task_ports INTERFACE)
foreach(module gimbal chassis shoot can dbus referee imu health log telemetry control_support navigation armored ui)
    target_include_directories(self_app_tasks_api INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/Modules/${module}")
    target_include_directories(self_app INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/Modules/${module}")
endforeach()
target_include_directories(self_app_tasks_api INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/Runtime" "${CMAKE_CURRENT_SOURCE_DIR}/Assembly")
# Source list and its dependencies are local to each module.
foreach(module gimbal chassis shoot can dbus referee imu health log telemetry control_support navigation armored ui)
    add_subdirectory(Modules/${module})
endforeach()
add_subdirectory(Runtime)
add_subdirectory(Assembly)
